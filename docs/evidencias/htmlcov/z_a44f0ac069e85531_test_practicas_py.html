34def test_cp204_el_robot_maneja_como_una_persona(): 

35 """CP-204: en la practica de manejo el robot pisa el embrague antes de cada cambio, mueve los pedales con la rapidez del panel (no salta) y pasa por 1a, 2a y 3a.""" 

36 m = practicas.Mando() 

37 e = practicas.Ejecucion(practicas.CATALOGO["normal"], m) 

38 historia = [] 

39 while not e.terminada: 

40 e.avanzar(DT) 

41 historia.append((m.v.marcha, m.v.embrague, m.v.acelerador, m.v.freno)) 

42 cambios = [i for i in range(1, len(historia)) if historia[i][0] != historia[i - 1][0]] 

43 assert cambios and all(historia[i - 1][1] >= 0.9 for i in cambios) 

44 saltos = max(abs(b[k] - a[k]) for a, b in zip(historia, historia[1:]) for k in (1, 2, 3)) 

45 assert saltos <= 7.0 * DT + 1e-9 

46 assert {"1", "2", "3"} <= {marcha for marcha, *_ in historia}
