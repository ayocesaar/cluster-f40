40def leer_crudo(ruta, cabecera, desde_el_final): 

41 """Bytes del registro n-esimo contando desde el final (1 = el ultimo), por acceso directo.""" 

42 secuencia = cabecera["total"] - desde_el_final 

43 desplazamiento = CABECERA.size + (secuencia % cabecera["capacidad"]) * REGISTRO.size 

44 fd = os.open(ruta, os.O_RDONLY | getattr(os, "O_BINARY", 0)) 

45 try: 

46 os.lseek(fd, desplazamiento, os.SEEK_SET) 

47 return desplazamiento, os.read(fd, REGISTRO.size) 

48 finally: 

49 os.close(fd)

52def ultimos(ruta, cabecera, n): 

53 registros, danados = [], 0 

54 for k in range(min(n, cabecera["total"], cabecera["capacidad"]), 0, -1): 

55 _, crudo = leer_crudo(ruta, cabecera, k) 

56 try: 

57 registros.append(desempaquetar(crudo)) 

58 except ValueError: 

59 danados += 1 

60 return registros, danados

67def describir_registro(ruta, cabecera, n): 

68 desplazamiento, crudo = leer_crudo(ruta, cabecera, n) 

69 lineas = [f"Registro {n} desde el final: byte {desplazamiento} = 64 + ({cabecera['total'] - n} mod " 

70 f"{cabecera['capacidad']}) x 32", f" hex: {hexadecimal(crudo)}"] 

71 for nombre, valor in zip(NOMBRES, REGISTRO.unpack(crudo)): 

72 lineas.append(f" {nombre:<38} {valor}") 

73 try: 

74 lineas.append(f" decodificado: {desempaquetar(crudo)}")
