1"""Pruebas de aceptacion del cluster dirigidas por datos. 

2 

3Lee los escenarios de un archivo CSV, los ejecuta sobre la interfaz Tkinter 

4real (con reloj simulado para que ciclos y parpadeos sean deterministas) y 

5compara lo que queda dibujado en pantalla con lo esperado en cada paso. 

6 

7Uso: python -m aceptacion.escenarios tests/datos/escenarios_aceptacion.csv 

8(requiere servidor grafico; en la VM: xvfb-run -a python -m aceptacion.escenarios ...)."""

69def _valor(accion, texto, linea): 

70 if accion == "-": 

71 return None 

72 if accion in DESLIZADORES: 

73 try: 

74 return float(texto) 

75 except ValueError: 

76 raise ErrorEscenarios(f"linea {linea}: el valor {texto!r} de {accion} no es numerico") from None 

77 if accion in INTERRUPTORES: 

78 if texto not in ("0", "1"): 

79 raise ErrorEscenarios(f"linea {linea}: {accion} solo acepta 0 o 1, no {texto!r}") 

80 return texto == "1" 

81 if texto not in DIRECCIONALES: 

82 raise ErrorEscenarios(f"linea {linea}: direccional debe ser OFF, IZQ o DER, no {texto!r}") 

83 return texto

86def _paso(fila, linea): 

87 fila = {c: (fila.get(c) or "").strip() for c in COLUMNAS} 

88 accion = fila["accion"] 

89 if accion != "-" and accion not in CONTROLES: 

90 raise ErrorEscenarios(f"linea {linea}: control desconocido {accion!r}") 

91 try: 

92 numero = int(fila["paso"]) 

93 avanzar_ms = float(fila["avanzar_ms"]) 

94 elemento = int(fila["elemento"]) 

95 except ValueError: 

96 raise ErrorEscenarios(f"linea {linea}: paso, avanzar_ms y elemento deben ser numericos") from None 

97 if avanzar_ms < 0: 

98 raise ErrorEscenarios(f"linea {linea}: avanzar_ms negativo: {avanzar_ms:g}") 

99 if not 1 <= elemento <= 10: 

100 raise ErrorEscenarios(f"linea {linea}: elemento fuera de 1-10: {elemento}") 

101 esperado = fila["esperado"] 

102 validos = ESPERADOS_LECTURA if elemento <= 5 else ESPERADOS_TESTIGO 

103 if esperado not in validos and not (elemento <= 5 and esperado.startswith("valor=")): 

104 raise ErrorEscenarios(f"linea {linea}: esperado {esperado!r} no valido para el elemento {elemento}") 

105 return Paso(linea, fila["escenario"], numero, accion, _valor(accion, fila["valor"], linea), 

106 avanzar_ms, elemento, esperado, fila["descripcion"])

109def cargar(ruta): 

110 """Lee el CSV y devuelve {escenario: [Paso, ...]} en el orden del archivo.""" 

111 ruta = pathlib.Path(ruta) 

112 if not ruta.is_file(): 

113 raise ErrorEscenarios(f"no existe el archivo de escenarios: {ruta}") 

114 escenarios = {} 

115 with ruta.open(encoding="utf-8", newline="") as archivo: 

116 lector = csv.DictReader(archivo) 

117 if lector.fieldnames is None: 

118 raise ErrorEscenarios(f"el archivo esta vacio: {ruta}") 

119 faltan = [c for c in COLUMNAS if c not in lector.fieldnames] 

120 if faltan: 

121 raise ErrorEscenarios(f"faltan columnas en {ruta}: {', '.join(faltan)}") 

122 for fila in lector: 

123 paso = _paso(fila, lector.line_num) 

124 escenarios.setdefault(paso.escenario, []).append(paso) 

125 if not escenarios: 

126 raise ErrorEscenarios(f"el archivo no tiene pasos: {ruta}") 

127 return escenarios

130def observar(tablero, elemento, esperado): 

131 """Lee del canvas lo que esta dibujado para el elemento indicado.""" 

132 it = tablero.items[elemento] 

133 if esperado.startswith("valor="): 

134 return "valor=" + tablero.canvas.itemcget(it["valor"], "text") 

135 color = tablero.canvas.itemcget(it["aguja" if elemento <= 5 else "forma"], "fill") 

136 return NOMBRES_COLOR.get(color, color)

139def ejecutar(pasos): 

140 """Ejecuta un escenario sobre una ventana nueva, en estado seguro inicial.""" 

141 raiz = tk.Tk() 

142 reloj = RelojSimulado() 

143 tablero = Tablero(raiz, EstadoCluster(reloj)) 

144 resultados = [] 

145 try: 

146 for paso in pasos: 

147 if paso.accion != "-": 

148 tablero.ajustar(CONTROLES[paso.accion], paso.valor) 

149 reloj.avanzar(paso.avanzar_ms) 

150 tablero.ciclo() 

151 observado = observar(tablero, paso.elemento, paso.esperado) 

152 resultados.append(Resultado(paso, observado, observado == paso.esperado)) 

153 finally: 

154 tablero.detener() 

155 raiz.destroy() 

156 return resultados

159def main(argv=None): 

160 argv = sys.argv[1:] if argv is None else argv 

161 if len(argv) != 1: 

162 print("uso: python -m aceptacion.escenarios ARCHIVO.csv", file=sys.stderr) 

163 return 2 

164 try: 

165 escenarios = cargar(argv[0]) 

166 except ErrorEscenarios as error: 

167 print(f"ERROR: {error}", file=sys.stderr) 

168 return 2 

169 

170 aprobados = 0 

171 for nombre, pasos in escenarios.items(): 

172 resultados = ejecutar(pasos) 

173 ok = all(r.aprobado for r in resultados) 

174 aprobados += ok 

175 print(f"{nombre}: {'APROBADO' if ok else 'FALLIDO'}") 

176 for r in resultados: 

177 p = r.paso 

178 valor = "-" if p.valor is None else p.valor 

179 print(f" {'OK' if r.aprobado else 'XX'} paso {p.numero}: {p.descripcion} | " 

180 f"{p.accion}={valor} +{p.avanzar_ms:g} ms | elemento {p.elemento}: " 

181 f"esperado={p.esperado} observado={r.observado}") 

182 print(f"Resumen: {aprobados}/{len(escenarios)} escenarios aprobados") 

183 return 0 if aprobados == len(escenarios) else 1
