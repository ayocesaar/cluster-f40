109def instalar_senales(app): 

110 """SIGTERM (kill PID) y SIGINT (Ctrl+C) producen un cierre ordenado (tema 3.2, DEF-15). 

111 

112 Sin esto, la accion por defecto de SIGTERM terminaba el proceso sin guardar el 

113 kilometraje ni registrar 'sistema fin'. El manejador solo agenda el cierre en el 

114 hilo de Tkinter: la interfaz no es segura entre hilos ni dentro de un manejador.""" 

115 def manejar(numero, _marco): 

116 app.bitacora.info("sistema", "senal_recibida", f"{signal.Signals(numero).name} cierre ordenado") 

117 app.raiz.after(0, app.cerrar) 

118 return {senal: signal.signal(senal, manejar) for senal in (signal.SIGTERM, signal.SIGINT)}
