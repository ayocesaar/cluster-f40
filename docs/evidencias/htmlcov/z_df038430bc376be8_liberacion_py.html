1"""Banco de liberacion de produccion (fin de linea) del F40 simulado. 

2 

3Antes de mandar un auto a produccion, y despues a cada unidad al final de la linea, se 

4corren pruebas normalizadas. Este banco las reproduce sobre el modelo fisico del panel 

5(panel/fisica.py) con la misma ECU (panel/ecu.py); las que tocan al cuadro usan el codigo 

6real del tablero (la conversion del lector, EstadoCluster y la presentacion). 

7 

8Normas (texto verificado; citas y limites en docs/realismo_y_liberacion.md): 

9- UN R13-H, anexo 3: frenado tipo 0 desde 100 km/h con el motor desembragado, 

10 s <= 0.1 v + 0.0060 v^2 y dm >= 6.43 m/s^2 (2.1.1 A); tipo I (fading, 1.5): 15 frenadas 

11 de v1 = 80 % vmax (maximo 120 km/h) a v1/2, una cada 45 s, a 3 m/s^2, en la marcha mas 

12 alta; en caliente >= 75 % de lo prescrito y >= 60 % de lo medido en frio. 

13 5.2.21.4: los testigos se encienden al energizar y solo se apagan si no hay falla. 

14- UN R39, 5.2.5 y 5.3 (velocimetro): para vmax > 150 km/h se prueba a 40, 80 y 120 km/h; 

15 la velocidad indicada nunca es menor que la real y 0 <= V1 - V2 <= 0.1 V2 + 4 km/h 

16 (+ 6 km/h en la conformidad de la produccion, anexo 3). 

17- Ciclo urbano ECE-15 de la prueba tipo I de emisiones: 195 s, 994.03 m, maximo 50 km/h. 

18Las cifras del vehiculo son aproximadas. Los valores de la Directiva 71/320/CEE vigentes en 

191987 quedan pendientes de verificar: se usan los de R13-H. El enfriamiento, el turbo y el 

20interbloqueo de puertas son criterios del equipo, no de una norma."""

46def desaceleracion_media(vb, ve, sb, se): 

47 """Desaceleracion media totalmente desarrollada (m/s^2): (vb^2 - ve^2) / (25.92 (se - sb)), 

48 con vb = 0.8 v0 y ve = 0.1 v0 en km/h y sb, se en metros.""" 

49 return (vb ** 2 - ve ** 2) / (25.92 * (se - sb))

78def interpolar(perfil, t): 

79 for (t0, v0), (t1, v1) in zip(perfil, perfil[1:]): 

80 if t0 <= t <= t1: 

81 return v0 + (v1 - v0) * (t - t0) / (t1 - t0) if t1 > t0 else v1 

82 return perfil[-1][1]

96class Conductor: 

97 """Robot conductor del banco: sigue un perfil de velocidad con acelerador y freno. 

98 

99 Calcula la aceleracion que necesita (pendiente del perfil mas una correccion proporcional 

100 al error) y la convierte en pedal con el mismo modelo del vehiculo.""" 

101 

102 def __init__(self, perfil, elegir_marcha=None, anticipacion_s=0.6, ganancia=1.2): 

103 self.perfil = perfil 

104 self.elegir_marcha = elegir_marcha 

105 self.anticipacion_s = anticipacion_s 

106 self.ganancia = ganancia 

107 

108 def objetivo(self, t): 

109 return self.perfil(t) if callable(self.perfil) else interpolar(self.perfil, t) 

110 

111 def aplicar(self, v, t): 

112 objetivo = self.objetivo(t) 

113 if self.elegir_marcha is not None: 

114 marcha = self.elegir_marcha(v, objetivo) 

115 if marcha != v.marcha: 

116 try: 

117 v.cambiar(marcha) 

118 except ValueError: 

119 pass 

120 v.embrague = 0.0 

121 pendiente = (self.objetivo(t + self.anticipacion_s) - objetivo) / self.anticipacion_s 

122 deseada = pendiente + self.ganancia * (objetivo - v.velocidad) # km/h por segundo 

123 necesario = deseada + resistencia_al_avance(v.velocidad) 

124 acelerador = freno = 0.0 

125 if v.acoplado and necesario > 0: 

126 acelerador = necesario / max(1.0, v.empuje_maximo()) # el par cambia con rpm y turbo 

127 else: 

128 ayuda = FRENO_MOTOR if v.acoplado else 0.0 

129 if -necesario > ayuda: 

130 freno = (-necesario - ayuda) / (FRENADO * max(0.05, v.eficiencia_frenos)) 

131 v.acelerador = max(0.0, min(1.0, acelerador)) 

132 v.freno = max(0.0, min(1.0, freno)) 

133 return objetivo

136class CadenaCuadro: 

137 """El cuadro real sin ventana: cada lectura pasa por el formato del archivo y la validacion 

138 del lector (formatear -> convertir) y llega a EstadoCluster, que avanza un ciclo de 100 ms.""" 

139 

140 CICLO_MS = 100 

141 

142 def __init__(self, odometro_km=0.0): 

143 self.reloj = RelojSimulado() 

144 self.estado = EstadoCluster(self.reloj, odometro_km)

146 def aplicar(self, lecturas, extras=None, con_vista=True): 

147 for clave, valor in {**lecturas, **(extras or {})}.items(): 

148 setter = (CLAVES.get(clave) or CLAVES_EXTRA[clave])[0] 

149 getattr(self.estado, setter)(convertir(clave, formatear(clave, valor), RANGOS_ESPECIFICACION)) 

150 self.reloj.avanzar(self.CICLO_MS) 

151 self.estado.tick() 

152 return (vista(self.estado), vista_extra(self.estado)) if con_vista else None

228def _frenada_tipo0(v, v0=V_TIPO0): 

229 """Frenada a fondo con el motor desembragado; devuelve (distancia m, dm m/s^2).""" 

230 v.cambiar("N") 

231 v.velocidad, v.acelerador, v.freno = v0, 0.0, 1.0 

232 vb, ve = 0.8 * v0, 0.1 * v0 

233 s, sb, se = 0.0, None, None 

234 for _ in range(int(60 / DT)): 234 ↛ 245line 234 didn't jump to line 245 because the loop on line 234 didn't complete

235 antes = v.velocidad 

236 v.avanzar(DT) 

237 tramo = (antes + v.velocidad) / 2 / 3.6 * DT 

238 for limite, nombre in ((vb, "sb"), (ve, "se")): 

239 if (sb if nombre == "sb" else se) is None and antes > limite >= v.velocidad: 

240 cruce = s + tramo * (antes - limite) / (antes - v.velocidad) 

241 sb, se = (cruce, se) if nombre == "sb" else (sb, cruce) 

242 s += tramo 

243 if v.velocidad <= 0.0: 

244 break 

245 v.freno = 0.0 

246 return s, desaceleracion_media(vb, ve, sb, se)

257def prueba_tipo1(fallas): 

258 _, dm_frio = _frenada_tipo0(_vehiculo(fallas)) 

259 v1, v2 = velocidades_tipo1(VMAX) 

260 v = _vehiculo(fallas, marcha="5", velocidad=v1) 

261 for ciclo in range(1, TIPO1_CICLOS + 1): 

262 t = 0.0 

263 while t < TIPO1_CICLO_S: 

264 if t < 1e-9 or v.freno > 0 and v.velocidad > v2: # frenada de v1 a v2 a 3 m/s^2 

265 v.acelerador = 0.0 

266 falta = TIPO1_DECEL * 3.6 - resistencia_al_avance(v.velocidad) - FRENO_MOTOR 

267 v.freno = max(0.01, min(1.0, falta / (FRENADO * max(0.05, v.eficiencia_frenos)))) 

268 elif ciclo == TIPO1_CICLOS: # tras la ultima frenada: prueba en caliente 

269 break 

270 else: # volver a v1 y sostenerla 

271 v.freno = 0.0 

272 v.acelerador = 1.0 if v.velocidad < v1 else resistencia_al_avance(v1) / max(1.0, v.empuje_maximo()) 

273 v.avanzar(DT) 

274 t += DT 

275 caliente = v.temp_frenos 

276 _, dm_caliente = _frenada_tipo0(v) 

277 minimo = max(TIPO1_PRESCRITO * DM_TIPO0, TIPO1_MEDIDO * dm_frio) 

278 return Resultado("L04", "Frenado tipo I (perdida de eficiencia en caliente)", "UN R13-H anexo 3, 1.5", 

279 f"{TIPO1_CICLOS} frenadas {v1:g}->{v2:g} km/h cada {TIPO1_CICLO_S:g} s; en caliente dm >= " 

280 f"75 % de {DM_TIPO0:g} y >= 60 % del valor en frio", 

281 f"frenos a {caliente:.0f} C; dm caliente {dm_caliente:.2f} m/s2 (frio {dm_frio:.2f}; minimo " 

282 f"{minimo:.2f})", dm_caliente >= minimo)

306def recorrer_ece15(fallas=None, odometro_km=0.0): 

307 """Conduce el ciclo urbano completo; devuelve (vehiculo, cuadro, error maximo, km reales, avisos).""" 

308 v, ecu, avisos = _vehiculo(fallas), Ecu(), [] 

309 v.temp_agua = 40.0 # el ciclo empieza con el motor frio 

310 conductor, cuadro = Conductor(ECE15, marcha_ece15), CadenaCuadro(odometro_km) 

311 error, km, t, n = 0.0, 0.0, 0.0, 0 

312 while t < DURACION_ECE15_S - 1e-9: 

313 conductor.aplicar(v, t) 

314 _paso(v, ecu, avisos) 

315 t, n = t + DT, n + 1 

316 km += v.velocidad * DT / 3600 

317 error = max(error, abs(v.velocidad - conductor.objetivo(t))) 

318 if n % 2 == 0: 

319 cuadro.aplicar(v.lecturas(), v.lecturas_extra(ecu.proteccion), con_vista=False) 

320 return v, cuadro, error, km, avisos
