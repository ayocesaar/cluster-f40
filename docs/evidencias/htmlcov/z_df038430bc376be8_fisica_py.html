175 def avanzar(self, dt): 

176 """Avanza dt segundos (se recomienda 0.05).""" 

177 if dt <= 0: 177 ↛ 178line 177 didn't jump to line 178 because the condition on line 177 was never true

178 return 

179 self.tiempo_s += dt 

180 mitad = 0.5 if self.fallas.ecu_banco else 1.0 # un banco sin encendido: medio motor 

181 # Con la traccion bloqueada por la ECU (puerta abierta) el acelerador no tiene efecto. 

182 acelerador = 0.0 if self.traccion_bloqueada else _limitar(self.acelerador, 0.0, 1.0) 

183 freno = _limitar(self.freno, 0.0, 1.0) 

184 

185 if not self.motor: 

186 self.rpm = max(0.0, self.rpm - 4000 * dt) 

187 empuje = 0.0 

188 self.patinando = False 

189 elif self.acoplado: 

190 relacion = CORTE / VELOCIDAD_AL_CORTE[self.marcha] # rpm por km/h 

191 corte = self.rpm >= LIMITADOR - 1 # el limitador corta la inyeccion 

192 empuje = 0.0 if corte else acelerador * EMPUJE[self.marcha] * mitad * self.factor_par 

193 limite = TRACCION_MAXIMA * self.adherencia 

194 self.patinando = empuje > limite # sin control de traccion 

195 if self.patinando: 

196 empuje = limite * DESLIZAMIENTO 

197 if acelerador == 0: 

198 empuje -= FRENO_MOTOR 

199 else: 

200 self.patinando = False 

201 objetivo = RALENTI + acelerador * (LIMITADOR - RALENTI) # revoluciones en vacio 

202 if self.fallas.ecu_banco: # ralenti irregular 

203 objetivo += RALENTI_IRREGULAR * math.sin(2 * math.pi * 1.3 * self.tiempo_s) * (1 - acelerador) 

204 paso = (6000 if objetivo > self.rpm else 3000) * dt 

205 self.rpm += _limitar(objetivo - self.rpm, -paso, paso) 

206 empuje = 0.0 

207 

208 resistencia = resistencia_al_avance(self.velocidad) 

209 frenado = FRENADO * freno * self.eficiencia_frenos 

210 limite_freno = FRENADO_ADHERENCIA * self.adherencia 

211 self.bloqueado = frenado > limite_freno and self.velocidad > 1.0 # sin ABS 

212 if self.bloqueado: 

213 frenado = limite_freno * DESLIZAMIENTO 

214 decel = resistencia + frenado 

215 antes = self.velocidad 

216 self.velocidad = max(0.0, self.velocidad + (empuje - decel) * dt) 

217 if self.motor and self.acoplado: 

218 self.velocidad = min(self.velocidad, LIMITADOR / relacion) 

219 self.rpm = max(RALENTI, self.velocidad * relacion) 

220 self._temperatura_frenos(frenado, antes, dt) 

221 

222 # Turbo: sigue al acelerador con retraso y solo da presion por encima de 2500 rpm. 

223 carga = acelerador * _limitar((self.rpm - 2500) / 2500, 0.0, 1.0) if self.motor else 0.0 

224 maximo = TURBO_MAXIMO * (1.65 if self.fallas.sobrepresion_turbo else 1.0) * (0.55 if mitad < 1 else 1.0) 

225 objetivo_turbo = maximo * carga 

226 self.presion_turbo += (objetivo_turbo - self.presion_turbo) * min(1.0, dt / TURBO_RETRASO_S) 

227 

228 # Temperatura: se acerca a la de operacion con el motor encendido y se enfria apagado. 

229 carga_motor = acelerador * _limitar(self.rpm / 7000.0, 0.0, 1.0) if self.motor else 0.0 

230 if self.motor: 

231 objetivo_t = (TEMPERATURA_SOBRECALENTADO if self.fallas.radiador 

232 else TEMPERATURA_NORMAL + CARGA_TERMICA * carga_motor 

233 - (ENFRIAMIENTO_VENTILADOR if self.ventilador else 0.0)) 

234 else: 

235 objetivo_t = 40.0 

236 self.temp_agua += (objetivo_t - self.temp_agua) * min(1.0, dt / CALENTAMIENTO_S) 

237 

238 # Aceite: sigue al agua mas lento y se calienta con la carga; con poco aceite, mas. 

239 if self.motor: 

240 objetivo_aceite = (self.temp_agua + ACEITE_SOBRE_AGUA + ACEITE_CARGA * carga_motor 

241 + (15.0 if self.nivel_aceite < NIVEL_ACEITE_CRITICO else 0.0)) 

242 else: 

243 objetivo_aceite = 40.0 

244 self.temp_aceite += (objetivo_aceite - self.temp_aceite) * min(1.0, dt / ACEITE_CONSTANTE_S) 

245 

246 # Ventilador del radiador con histeresis. 

247 if self.motor and self.temp_agua >= VENTILADOR_ENCIENDE: 

248 self.ventilador = True 

249 elif not self.motor or self.temp_agua <= VENTILADOR_APAGA: 

250 self.ventilador = False 

251 # Bateria: el alternador carga lo que sobra de las cargas; si no carga, la bateria las paga. 

252 cargas, generacion = self._corrientes() 

253 cambio = (generacion - cargas) * dt * ACELERACION_ELECTRICA / 3600.0 / CAPACIDAD_BATERIA_AH 

254 self.bateria = _limitar(self.bateria + cambio, 0.0, 1.0) 

255 

256 # Combustible: sin combustible el motor se apaga solo. 

257 if self.motor: 

258 consumo_lh = CONSUMO_RALENTI_LH + CONSUMO_PLENA_LH * carga_motor 

259 self.combustible_l = max(0.0, self.combustible_l - consumo_lh * dt / 3600.0) 

260 if self.combustible_l == 0.0: 

261 self.motor = False 

262 # Aceite: la fuga vacia el deposito (mas rapido con el motor en marcha, por la presion). 

263 if self.fallas.fuga_aceite: 

264 ritmo = FUGA_ACEITE_PCT_S if self.motor else FUGA_ACEITE_PCT_S / 10 

265 self.nivel_aceite = max(0.0, self.nivel_aceite - ritmo * dt)

308 def _temperatura_frenos(self, frenado_kmh_s, antes_kmh, dt): 

309 """Los frenos convierten la energia del auto en calor y se enfrian con el aire (mas rapido al avanzar).""" 

310 media_ms = (antes_kmh + self.velocidad) / 2 / 3.6 

311 energia = MASA_KG * (frenado_kmh_s / 3.6) * media_ms * dt 

312 energia = min(energia, 0.5 * MASA_KG * (antes_kmh / 3.6) ** 2) 

313 self.temp_frenos += energia / CALOR_FRENOS_J_K 

314 enfriamiento = (0.001 + 0.00008 * media_ms) * dt 

315 self.temp_frenos -= (self.temp_frenos - AMBIENTE_C) * min(1.0, enfriamiento)
